prototype_controller.ino (Hardware Interrupt & Solenoid Pulse) 
// Yards Brewing Co. - Pre-Fill Can Inspection Prototype Controller
// Synchronizes M12 optical trigger, Keyence / ELP camera, and pneumatic rejector

#include <Arduino.h>

const int PIN_PHOTO_TRIGGER = 2;   // M12 Diffuse optical sensor (Step 3)
const int PIN_CAM_TRIGGER   = 3;   // 24V Opto-isolated pulse to Keyence VS-C320MX
const int PIN_CV_DECISION   = 4;   // Input from Keyence Out1 (HIGH = REJECT)
const int PIN_AIR_SOLENOID  = 5;   // MOSFET driver for 80 PSI Air Shot Valve
const int PIN_SERVO_PWM     = 6;   // Desktop hobby servo kicker alternative

// Calibration Constants (10 inches/sec @ 150mm spacing)
const unsigned long TRANSIT_DELAY_MS   = 572; // 590ms transit - 18ms camera latency
const unsigned long SOLENOID_PULSE_MS  = 40;  // High-velocity 40ms air shot

struct CanEvent {
  unsigned long fireTime;
  bool reject;
};

#define QUEUE_SIZE 8
CanEvent eventQueue[QUEUE_SIZE];
volatile int qHead = 0;
volatile int qTail = 0;

void IRAM_ATTR onCanDetected() {
  // 1. Immediate trigger pulse to camera to freeze motion (1/2000s exposure)
  digitalWrite(PIN_CAM_TRIGGER, HIGH);
  delayMicroseconds(100);
  digitalWrite(PIN_CAM_TRIGGER, LOW);
}

void setup() {
  pinMode(PIN_PHOTO_TRIGGER, INPUT_PULLUP);
  pinMode(PIN_CAM_TRIGGER, OUTPUT);
  pinMode(PIN_CV_DECISION, INPUT);
  pinMode(PIN_AIR_SOLENOID, OUTPUT);
  attachInterrupt(digitalPinToInterrupt(PIN_PHOTO_TRIGGER), onCanDetected, RISING);
  Serial.begin(115200);
}

void loop() {
  // Check if Keyence flagged reject for current frame
  if (digitalRead(PIN_CV_DECISION) == HIGH) {
    unsigned long targetFire = millis() + TRANSIT_DELAY_MS;
    eventQueue[qHead] = { targetFire, true };
    qHead = (qHead + 1) % QUEUE_SIZE;
  }

  // Fire air shot when can arrives at nozzle
  if (qHead != qTail && millis() >= eventQueue[qTail].fireTime) {
    if (eventQueue[qTail].reject) {
      digitalWrite(PIN_AIR_SOLENOID, HIGH);
      delay(SOLENOID_PULSE_MS);
      digitalWrite(PIN_AIR_SOLENOID, LOW);
    }
    qTail = (qTail + 1) % QUEUE_SIZE;
  }
}
